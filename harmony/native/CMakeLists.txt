cmake_minimum_required(VERSION 3.16)
project(madcore_napi LANGUAGES CXX)
if(NOT DEFINED MADCORE_LIBRARY OR NOT EXISTS "${MADCORE_LIBRARY}")
    message(FATAL_ERROR "MADCORE_LIBRARY must name the separately built arm64 libmadcore.so")
endif()
add_library(madcore SHARED IMPORTED)
set_target_properties(madcore PROPERTIES IMPORTED_LOCATION "${MADCORE_LIBRARY}")
add_library(madcore_napi SHARED madcore_napi.cpp)
target_compile_features(madcore_napi PRIVATE cxx_std_17)
target_compile_options(madcore_napi PRIVATE -Wall -Wextra -Werror)
target_link_libraries(madcore_napi PRIVATE madcore libace_napi.z.so)
